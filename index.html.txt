  <div class="container">
    <!-- نموذج تسجيل الدخول وإنشاء الحساب -->
    <div id="auth-section" class="card">
      <h2>تسجيل الدخول / إنشاء حساب جديد</h2>
      <input type="text" id="username" placeholder="اسم المستخدم" />
      <input type="password" id="password" placeholder="كلمة المرور" />
      <button id="login-btn" onclick="login()">تسجيل الدخول</button>
      <button id="register-btn" onclick="register()" style="background-color: #42b72a; margin-top: 5px;">إنشاء حساب جديد</button>
      <p id="auth-error" style="color: red; margin-top: 10px;"></p>
    </div>

    <!-- واجهة المنصة بعد الدخول -->
    <div id="app-section" class="hidden">
      <!-- إنشاء منشور -->
      <div class="card">
        <h3>شارك ما تفكر فيه</h3>
        <textarea id="post-content" rows="3" placeholder="اكتب منشوراً هنا..."></textarea>
        <button onclick="createPost()">نشر</button>
      </div>

      <!-- غرفة الشات المباشر -->
      <div class="card">
        <h3>المحادثة الفورية المباشرة 💬</h3>
        <div id="chat-box" class="chat-box"></div>
        <div style="display: flex; gap: 5px;">
          <input type="text" id="chat-msg" placeholder="اكتب رسالتك..." style="margin:0;" />
          <button onclick="sendMessage()" style="width: 100px; margin:0;">إرسال</button>
        </div>
      </div>

      <!-- تغذية المنشورات -->
      <div class="card">
        <h3>خلاصة الأخبار (Feed)</h3>
        <div id="posts-feed"></div>
      </div>

  <script>
    // افتح socket مع السماح بتمرير الكوكيز (لـ session)
    const socket = io({ withCredentials: true });
    let currentUser = '';

    // helper لتنسيق التاريخ (ISO → عرض عربي)
    function formatDate(iso) {
      try {
        const d = new Date(iso);
        return d.toLocaleString('ar-EG');
      } catch (e) {
        return iso;
      }
    }

    // التحقق عند فتح الصفحة (أرسل الكوكيز)
    async function checkAuth() {
      const res = await fetch('/api/me', { credentials: 'include' });
      if (!res.ok) return;
      const data = await res.json();
      if (data.loggedIn) {
        setupApp(data.username);
      }
    }

    function setupApp(username) {
      currentUser = username;
      document.getElementById('user-display').innerText = 'أهلاً، ' + username;
      document.getElementById('auth-section').classList.add('hidden');
      document.getElementById('app-section').classList.remove('hidden');
      loadPosts();
    }

    async function register() {
      const u = document.getElementById('username').value.trim();
      const p = document.getElementById('password').value;
      const res = await fetch('/api/register', {
        method: 'POST',
        credentials: 'include',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ username: u, password: p })
      });
      const data = await res.json();
      if (res.ok && data.success) {
        setupApp(data.username);
      } else {
        document.getElementById('auth-error').innerText = data.error || 'خطأ أثناء التسجيل';
      }
    }

    async function login() {
      const u = document.getElementById('username').value.trim();
      const p = document.getElementById('password').value;
      const res = await fetch('/api/login', {
        method: 'POST',
        credentials: 'include',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ username: u, password: p })
      });
      const data = await res.json();
      if (res.ok && data.success) {
        setupApp(data.username);
      } else {
        document.getElementById('auth-error').innerText = data.error || 'خطأ أثناء الدخول';
      }
    }

    async function createPost() {
      const content = document.getElementById('post-content').value.trim();
      if (!content) return;
      const res = await fetch('/api/posts', {
        method: 'POST',
        credentials: 'include',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ content })
      });
      if (res.ok) {
        document.getElementById('post-content').value = '';
        loadPosts();
      } else {
        const err = await res.json().catch(() => ({}));
        alert(err.error || 'خطأ عند إنشاء المنشور');
      }
    }

    async function loadPosts() {
      const res = await fetch('/api/posts', { credentials: 'include' });
      if (!res.ok) return;
      const posts = await res.json();
      const feed = document.getElementById('posts-feed');
      feed.innerHTML = posts.map(p => {
        const time = p.createdAt ? formatDate(p.createdAt) : '';
        return `
          <div class="post">
            <div><span class="post-author">${escapeHtml(p.author || '')}</span> <span class="post-time">${time}</span></div>
            <p>${escapeHtml(p.content || '')}</p>
          </div>
        `;
      }).join('');
    }

    // إرسال رسالة: لا ترسل اسم المرسل من العميل (تعتمد الجلسة على الخادم)
    function sendMessage() {
      const input = document.getElementById('chat-msg');
      const text = input.value.trim();
      if (!text) return;
      socket.emit('send_message', { text });
      input.value = '';
    }

    socket.on('receive_message', (msg) => {
      appendMsg(msg);
    });

    socket.on('chat_history', (history) => {
      const box = document.getElementById('chat-box');
      box.innerHTML = '';
      history.forEach(appendMsg);
    });

    function appendMsg(msg) {
      const box = document.getElementById('chat-box');
      const div = document.createElement('div');
      const isMe = msg.sender === currentUser;
      div.className = 'msg ' + (isMe ? 'me' : 'other');
      const time = msg.createdAt ? formatDate(msg.createdAt) : (msg.time || '');
      div.innerHTML = `<strong>${isMe ? 'أنت' : escapeHtml(msg.sender || 'مجهول')}:</strong> ${escapeHtml(msg.text || '')} <div class="post-time">${escapeHtml(time)}</div>`;
      box.appendChild(div);
      box.scrollTop = box.scrollHeight;
    }

    // Basic escaping to reduce XSS when inserting text into HTML
    function escapeHtml(str) {
      return String(str)
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;')
        .replace(/'/g, '&#039;');
    }

    checkAuth();
  </script>
